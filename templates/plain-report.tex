% Academic section numbering, driven by `section_numbering:` in report.yml.
% With the key absent the renderer derives the default from the confirmed
% route (#23): numbered on the academic route (every report written before
% `route:` existed), unnumbered on the non-academic routes the routing
% contract forbids from auto-including academic numbering. An explicit key
% overrides the route default in either direction.
%
% Lowering secnumdepth suppresses only the printed label. The headings are
% still \section, so \@sect keeps calling \addcontentsline (table of contents
% and list-of-figures machinery intact) and hyperref keeps placing its anchor.
% Starring the headings would have dropped both.
\newcommand{\reportsectionnumbering}{{{SECTION_NUMBERING}}}
\ifthenelse{\equal{\reportsectionnumbering}{false}}{\setcounter{secnumdepth}{0}}{}
