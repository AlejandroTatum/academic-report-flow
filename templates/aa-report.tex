% AA (Aprendizaje Autónomo) report template: the approved editorial design
% (proposal AA-B, aa-acd-templates S1/S2).
%   * A4 portrait, no cover page. ONLY PAGE 1 IS CREATIVE: an institutional
%     letterhead (faculty logo, university, faculty, career), the
%     identification "ficha" hanging in the left margin, then the kicker
%     "ACTIVIDAD AA-N · ASIGNATURA", the large serif title and the report title
%     as an italic subtitle. The opening text wraps beside the ficha.
%   * From the end of the ficha on, and on every later page, the layout is a
%     conventional full-width text block with inline section numbers, a
%     discreet running header (pages 2+) and a "page / total" footer.
%   * Body Source Serif 4 10.5pt, headings and labels Source Sans 3, code
%     Source Code Pro, crest green (#1F4E3D) as the only accent, grey rules.
%   * The ficha is generated from metadata (see aa_ficha in the renderer); the
%     fixed sections are authored in body.md. The bibliography is numbered like
%     any section and the annexes render after it through the renderer's
%     after-bibliography slot. Unlike the institutional templates
%     (onehalfspacing), the report keeps the mockup's tighter 1.12 line spacing.
%
% The ficha zone (S2). The ficha is drawn in the page-1 background and its
% height is measured once. Three rules keep the text clear of it, whatever the
% length of the first section; none needs anything from the author:
%   1. Prose, headings and list items that START beside the ficha get a
%      \parshape: as many indented lines as fit before the end of the ficha,
%      then full width. The count comes from the real position on the page.
%   2. Block material that would START beside it (tables, code, display math,
%      boxed answers, a [H] float) is pushed below the ficha by \zoneclear,
%      which the hooks below add before each such environment.
%   3. Floating figures and tables cannot take the top of page 1 (that is where
%      the ficha is): \suppressfloats[t] on page 1. They go to the bottom of
%      page 1 or to the next page, which are both clear of the ficha.
% The zone closes for good at the first paragraph that starts below the ficha
% or on page 2, so the change of width never meets a page break.

% ---------------------------------------------------------------------------
% The ficha zone: paragraphs that start beside the ficha are narrowed, block
% material is pushed below it. See the header of this file for the three rules.
% ---------------------------------------------------------------------------
\makeatletter
\newlength{\zoneX}\setlength{\zoneX}{3.8cm}       % 6.5cm - left margin 2.7cm
\newlength{\zonedrop}\setlength{\zonedrop}{1.1cm} % letterhead space above the title
\newlength{\zonebottom}
\newsavebox{\fichabox}
\newcount\zoneon \zoneon=\z@                       % 1 = zone open (set once the ficha is measured)
\newdimen\zone@L \newdimen\zone@W \newcount\zone@n
% Number of baselines that fit between the current position and the end of the
% zone; the paragraph gets that many indented lines, then full width. Rule 1.
\def\zone@shape{%
  % Inside lists \item has already set the paragraph shape from these two.
  \zone@L=\@totalleftmargin \zone@W=\linewidth
  % +4pt: a line box starts ~3.5pt above its glyphs, so lines that begin
  % within 4pt of the end of the ficha are still narrow.
  \@tempdima=\dimexpr\zonebottom-\pagetotal+4pt\relax
  \ifdim\@tempdima>\z@
    % Round UP: a line that starts above the zone's end must stay narrow.
    \advance\@tempdima by\baselineskip \advance\@tempdima by-1sp
    \zone@n=\@tempdima \@tempcntb=\baselineskip
    \divide\zone@n by\@tempcntb
    % Never narrower than the page can hold: a nearly full page (a bottom
    % float already took the room) must not carry narrow lines over its break.
    \ifdim\pagegoal<\maxdimen
      \@tempdimb=\dimexpr\pagegoal-\pagetotal\relax
      \ifdim\@tempdimb<\z@ \@tempdimb=\z@ \fi
      \@tempcnta=\@tempdimb \divide\@tempcnta by\@tempcntb
      \ifnum\zone@n>\@tempcnta \zone@n=\@tempcnta \fi
    \fi
    \ifnum\zone@n>\z@
      \edef\zone@spec{\parshape \the\numexpr\zone@n+1\relax}%
      \@tempcnta=\z@
      \@whilenum\@tempcnta<\zone@n\do{%
        \edef\zone@spec{\zone@spec\space\the\dimexpr\zone@L+\zoneX\relax
                        \space\the\dimexpr\zone@W-\zoneX\relax}%
        \advance\@tempcnta\@ne}%
      \edef\zone@spec{\zone@spec\space\the\zone@L\space\the\zone@W}%
      \zone@spec
    \fi
  \fi}
\def\zone@check{%
  \ifnum\zoneon>\z@
    \ifinner\else
      \ifnum\c@page=\@ne
        \ifdim\pagetotal<\zonebottom \zone@shape \else \global\zoneon=\z@ \fi
      \else
        \global\zoneon=\z@
      \fi
    \fi
  \fi}
\AddToHook{para/begin}{\zone@check}
% Rule 2: push block material below the ficha. Only in vertical mode: inline
% material (an inline tabular, math inside a paragraph) is part of a shaped
% paragraph and stays beside the ficha, inside the narrow lines.
% +8pt: \zonebottom stops 8pt short of the ficha box (room for the text's line
% box), but a display set right after a heading with no indent has no empty
% line above it, so a wide one would touch the ficha's last rule.
\newcommand{\zoneclear}{\ifvmode\ifnum\zoneon>\z@\ifinner\else\ifnum\c@page=\@ne
  \ifdim\pagetotal<\dimexpr\zonebottom+8pt\relax \par\vskip\dimexpr\zonebottom-\pagetotal+8pt\relax\fi\fi\fi\fi\fi}
\newcommand{\zone@blocks}{longtable,xltabular,tabular,tabularx,lstlisting,verbatim,%
  equation,equation*,align,align*,gather,gather*,multline,multline*,flalign,flalign*,%
  alignat,alignat*,displaymath,eqnarray,eqnarray*,answerbox,tikzpicture,%
  figure,figure*,table,table*}
\@for\zone@env:=\zone@blocks\do{%
  \edef\zone@hook{env/\zone@env/before}%
  \expandafter\AddToHook\expandafter{\zone@hook}{\zoneclear}}

\makeatother
