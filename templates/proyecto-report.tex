% Proyecto report template (format: proyecto, odd/tasks/project-format.md S2-S4).
% The AA template's look (aa-report.tex: UNL letterhead with logo, faculty and
% career lines, Source Serif 4 / Source Sans 3 / Source Code Pro, crest green,
% numbered green headings, booktabs tables) for a project report:
%   * page 1: the letterhead, the eyebrow "PROYECTO · ASIGNATURA", the large
%     serif title "Proyecto" and the report title as an italic subtitle;
%   * no sidebar: the identification data is a normal full-width table in the
%     flow (see proyecto_identification_table in the renderer), right under
%     the title block and before the first section; the body uses the full
%     text width on every page;
%   * a heading never parts from the start of its content and a short table
%     is never split (the renderer ties them with \nopagebreak, see
%     keep_blocks_together; no club or widow lines below).
% Like aa-report.tex, and unlike the institutional templates
% (onehalfspacing), it keeps the tighter 1.12 line spacing.

\documentclass[10pt,a4paper]{article}

\usepackage[fontsize=10.5pt]{fontsize}
% One geometry for every page: symmetric margins, no marginal column, the
% body at the full text width on every page (no sidebar).
\usepackage[a4paper,top=2.9cm,bottom=2.6cm,left=2.7cm,right=2.7cm,headheight=0.8cm,headsep=0.8cm,footskip=1.1cm]{geometry}
\usepackage{fontspec}
% Source Serif 4, Source Sans 3 and Source Code Pro ship with TeX Live and with
% the texlive/texlive image the renderer falls back to, but a minimal local
% install may lack them. Each face then degrades to a TeX Gyre face, the same
% fallbacks the other templates rely on, instead of failing to compile.
\IfFontExistsTF{Source Serif 4}{%
  \setmainfont{Source Serif 4}[Ligatures=TeX]
}{%
  \setmainfont{TeX Gyre Pagella}[Ligatures=TeX]
}
\IfFontExistsTF{Source Sans 3}{%
  \setsansfont{Source Sans 3}[Ligatures=TeX]
}{%
  \setsansfont{TeX Gyre Heros}[Ligatures=TeX]
}
\IfFontExistsTF{SourceSerif4-Light.otf}{%
  \newfontfamily\serlight{SourceSerif4}[Extension=.otf,UprightFont=*-Light,ItalicFont=*-LightIt]
}{%
  \newcommand{\serlight}{\normalfont}
}
\IfFontExistsTF{Source Code Pro}{%
  \setmonofont{Source Code Pro}[Scale=MatchLowercase]
}{%
  \setmonofont{TeX Gyre Cursor}[Scale=MatchLowercase]
}
% es-noindentfirst: babel-spanish indents the first paragraph after a heading by
% default; the mockup sets it flush and indents only the paragraphs that follow.

% Headings (S7): one style family, sans bold in green with the number inline
% in the same face; the levels differ only by size.
\titleformat{\section}{\sffamily\fontsize{13.5}{16}\selectfont\bfseries\color{accent}}{\thesection}{0.7em}{}
\titleformat{\subsection}{\sffamily\fontsize{11.5}{14}\selectfont\bfseries\color{accent}}{\thesubsection}{0.7em}{}

% Tables the Markdown converter emits for this template (booktabs: top, header
% and bottom rules, no vertical rules or shading), as in the mockup: sans body,
% bold header cells, ink rules.

% The paragraph right after a table is flush, as the mockup sets it: the table
% ends the thought that introduced it. A flag, not a group-local \everypar,
% because the converter closes its \begingroup right after the table.
% xltabular never runs cmd/endxltabular/after; its end command is the hook point.
\newif\ifreportflushnext
\AddToHook{cmd/endxltabular/after}{\global\reportflushnexttrue}
\AddToHook{para/begin}{\ifreportflushnext\global\reportflushnextfalse\kern-\parindent\fi}
